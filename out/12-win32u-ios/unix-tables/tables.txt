# Spec 12 clarifications 8 and 9: the unix libraries whose call tables are answered on iOS by
# stand-ins (nothing behind them: no CUPS, Samba, Kerberos or ntlm_auth). One per line:
#   <name> <header in the Wine 11.4 tree> <enum name> <count entry> [setup entry...]
# <name>_ios.c in this folder holds the two tables; gen-entries.sh writes <name>_ios_entries.h from
# the header. The setup entries return STATUS_SUCCESS, every other entry STATUS_NOT_SUPPORTED.
# Every <name> must also be listed (as its DLL file name) in out/09-ntdll-processes/unixlibs.txt.
# Spec 35 A4: opengl32's stand-in (clarification 8) is gone; its tables are opengl32's real unix
# side, libopengl32_unix.a of out/35-opengl-ios, which keeps clarification 8's answers while
# OpenGL is off (PG_OPENGL not 1).

netapi32  dlls/netapi32/unixlib.h      samba_funcs  unix_funcs_count  unix_netapi_init
kerberos  dlls/kerberos/unixlib.h      unix_funcs   unix_funcs_count  unix_process_attach
msv1_0    dlls/msv1_0/unixlib.h        ntlm_funcs   unix_funcs_count
