# Meson cross file for the i386 Windows half of DXMT on iOS (spec 06 Part A):
# 32-bit programs through WoW64. Target i686-w64-mingw32 with an llvm-mingw
# toolchain. Build with -Dwine_builtin_dll=true. DXMT_IOS is defined because
# the host system is Windows (meson.build).
# Replace @LLVM_MINGW_DIR@ with the toolchain directory (the one that holds bin/),
# for example: sed -i '' "s|@LLVM_MINGW_DIR@|$PWD/toolchains/llvm-mingw|" build-win-i686.txt

[constants]
toolchain = '@LLVM_MINGW_DIR@'
triple = 'i686-w64-mingw32'

[binaries]
c = toolchain / 'bin' / (triple + '-clang')
cpp = toolchain / 'bin' / (triple + '-clang++')
ar = toolchain / 'bin' / (triple + '-ar')
strip = toolchain / 'bin' / (triple + '-strip')
windres = toolchain / 'bin' / (triple + '-windres')

[properties]
needs_exe_wrapper = true

[host_machine]
system = 'windows'
cpu_family = 'x86'
cpu = 'i686'
endian = 'little'
